"use client";
import {useEffect,useMemo,useRef,useState} from "react";
import {BarChart3,CalendarDays,FileText,Settings2,Plus,Minus,Upload,Download,Save,Printer,ChevronDown,LogIn,LogOut,Clock3,Database,Send,Share2,Sparkles,X} from "lucide-react";
import {hrs,money,todayISO} from "@/lib/time";

const DEFAULT={hourlyEarning:20,defaultHours:8,otRate:30,minimumMinutes:20,workStart:"09:00",workEnd:"18:00",mode:"manual",offDayEarning:0,absentEarning:0,currency:"USD",autoCheckout:false};
const STATUSES=["Present","Off-day","Absent"];
const makeRow=(i,s,h)=>({id:Date.now()+i,name:"",employeeId:"",date:todayISO(),checkIn:"",checkOut:"",hours:h??s.defaultHours,otHours:0,hourlyEarning:s.hourlyEarning,status:"Present"});
const makeSheet=(n,s)=>({id:`sheet-${Date.now()}-${n}`,name:`Sheet ${n}`,sheetHours:s.defaultHours,rows:Array.from({length:12},(_,i)=>makeRow(i,s,s.defaultHours))});

export default function TimeSheetModule(){
 const [settings,setSettings]=useState(DEFAULT),[sheets,setSheets]=useState([]),[active,setActive]=useState(null),[tool,setTool]=useState(null),[notice,setNotice]=useState(""),input=useRef(null);
 useEffect(()=>{try{const a=JSON.parse(localStorage.getItem("sodah-ts-final-sheets")||"null"),b=JSON.parse(localStorage.getItem("sodah-ts-final-settings")||"null");if(b)setSettings({...DEFAULT,...b});if(a?.length){setSheets(a.slice(0,10));setActive(a[0].id)}else{const x=makeSheet(1,DEFAULT);setSheets([x]);setActive(x.id)}}catch{const x=makeSheet(1,DEFAULT);setSheets([x]);setActive(x.id)}},[]);
 useEffect(()=>{if(sheets.length)localStorage.setItem("sodah-ts-final-sheets",JSON.stringify(sheets))},[sheets]);
 useEffect(()=>localStorage.setItem("sodah-ts-final-settings",JSON.stringify(settings)),[settings]);
 const sheet=sheets.find(x=>x.id===active)||sheets[0];
 const setSheet=fn=>setSheets(a=>a.map(x=>x.id===sheet.id?fn(x):x));
 const flash=t=>{setNotice(t);setTimeout(()=>setNotice(""),2600)};
 const addSheet=()=>{if(sheets.length>=10){flash("Maximum of 10 sheets reached.");return}const x=makeSheet(sheets.length+1,settings);setSheets(a=>[...a,x]);setActive(x.id)};
 const deleteSheet=id=>{if(sheets.length===1)return flash("At least one sheet must remain.");const n=sheets.filter(x=>x.id!==id);setSheets(n);if(id===active)setActive(n[0].id)};
 const addRow=()=>setSheet(s=>({...s,rows:[...s.rows,makeRow(s.rows.length,settings,s.sheetHours)]}));
 const removeRow=()=>setSheet(s=>({...s,rows:s.rows.slice(0,-1)}));
 const addCol=()=>setSheet(s=>({...s,extra:[...(s.extra||[]),`Column ${(s.extra||[]).length+1}`]}));
 const removeCol=()=>setSheet(s=>({...s,extra:(s.extra||[]).slice(0,-1)}));
 const updateRow=(id,patch)=>setSheet(s=>({...s,rows:s.rows.map(r=>r.id===id?{...r,...patch}:r)}));
 const checkIn=id=>{const now=new Date().toTimeString().slice(0,5);updateRow(id,{date:todayISO(),checkIn:now,checkOut:"",status:"Present"});flash("Check-in recorded.")};
 const canCheckout=r=>{if(!r.checkIn||r.checkOut||r.status!=="Present")return false;const now=new Date();const [h,m]=r.checkIn.split(":").map(Number);const start=new Date();start.setHours(h,m,0,0);const min=new Date(start.getTime()+settings.minimumMinutes*60000);if(settings.mode==="manual")return now>=min;const [eh,em]=settings.workEnd.split(":").map(Number);const end=new Date();end.setHours(eh,em,0,0);const windowStart=new Date(end.getTime()-settings.minimumMinutes*60000);return now>=windowStart;};
 const checkOut=id=>{const r=sheet.rows.find(x=>x.id===id);if(!canCheckout(r)){flash(settings.mode==="automatic"?`Automatic check-out opens ${settings.minimumMinutes} minutes before ${settings.workEnd}.`:`Check-out opens after ${settings.minimumMinutes} minutes.`);return}updateRow(id,{checkOut:new Date().toTimeString().slice(0,5)});flash("Check-out recorded.")};
 const total=useMemo(()=>sheet?.rows.reduce((a,r)=>{const h=hrs(r.checkIn,r.checkOut),ot=Number(r.otHours||0),rate=Number(r.hourlyEarning||0),status=r.status||"Present";if(status!=="Present")return a;return {...a,h:a.h+h,ot:a.ot+ot,earn:a.earn+h*rate+ot*Number(settings.otRate||rate)}} ,{h:0,ot:0,earn:0})||{h:0,ot:0,earn:0},[sheet,settings]);
 const csv=()=>{const rows=[...["Check In / Out","Employee Name","Employee ID","Date","Hours","O.T","Hourly Earning","Worked Hours","O.T Earning","Total Earning","Status"],...sheet.rows.map(r=>[r.checkIn?"Checked in":"Check In",r.name,r.employeeId,r.date,r.hours,r.otHours,r.hourlyEarning,hrs(r.checkIn,r.checkOut),Number(r.otHours||0)*Number(settings.otRate||0),r.status])];const text=rows.map(a=>a.map(v=>`"${String(v??"").replaceAll('"','""')}"`).join(",")).join("\n");download(`${sheet.name}.csv`,new Blob([text],{type:"text/csv"}))};
 const xlsx=async()=>{const X=await import("xlsx");const data=sheet.rows.map(r=>({"Check In / Out":r.checkIn?"Checked in":"Check In","Employee Name":r.name,"Employee ID":r.employeeId,"Date":r.date,"Hours":r.hours,"O.T":r.otHours,"Hourly Earning":r.hourlyEarning,"Worked Hours":hrs(r.checkIn,r.checkOut),"O.T Earning":Number(r.otHours||0)*Number(settings.otRate||0),"Total Earning":hrs(r.checkIn,r.checkOut)*Number(r.hourlyEarning||0)+Number(r.otHours||0)*Number(settings.otRate||0),"Status":r.status}));const w=X.utils.book_new();X.utils.book_append_sheet(w,X.utils.json_to_sheet(data),sheet.name.slice(0,31));X.utils.book_append_sheet(w,X.utils.json_to_sheet(sheet.rows),"Raw Data");X.utils.book_append_sheet(w,X.utils.json_to_sheet([settings]),"Settings");X.writeFile(w,`${sheet.name}.xlsx`)};
 const download=(name,blob)=>{const u=URL.createObjectURL(blob),a=document.createElement("a");a.href=u;a.download=name;a.click();URL.revokeObjectURL(u)};
 const importExcel=async()=>{const file=input.current?.files?.[0];if(!file)return;const X=await import("xlsx"),w=X.read(await file.arrayBuffer()),d=X.utils.sheet_to_json(w.Sheets[w.SheetNames[0]],{defval:""});setSheet(s=>({...s,rows:d.map((r,i)=>({...makeRow(i,settings,s.sheetHours),name:r["Employee Name"]||"",employeeId:r["Employee ID"]||"",date:r.Date||todayISO(),checkIn:r["Check In"]||"",checkOut:r["Check Out"]||"",hours:Number(r.Hours||s.sheetHours),otHours:Number(r["O.T"]||0),hourlyEarning:Number(r["Hourly Earning"]||settings.hourlyEarning),status:r.Status||"Present"}))}));input.current.value="";flash("Sheet imported.")};
 if(!sheet)return <div className="p-10">Loading Time Sheet…</div>;
 return <main className="min-h-screen bg-[#f7f9f8]">
  <TopBar tool={tool} setTool={setTool} addSheet={addSheet} count={sheets.length} addRow={addRow} removeRow={removeRow} addCol={addCol} removeCol={removeCol} csv={csv} xlsx={xlsx} importExcel={()=>input.current?.click()} />
  <input ref={input} type="file" accept=".xlsx,.xls,.csv" onChange={importExcel} className="hidden"/>
  {notice&&<div className="noprint mx-auto max-w-[1800px] px-4 pt-3"><div className="rounded-lg bg-emerald-50 px-3 py-2 text-xs font-bold text-emerald-700">{notice}</div></div>}
  <div className="mx-auto max-w-[1800px] p-3 sm:p-4">
   <div className="noprint mb-3 flex flex-wrap items-center gap-3"><div><h1 className="text-lg font-black">Employee Time Sheet</h1><p className="text-xs text-slate-500">Excel-style sheet • editable • scrollable • up to 10 sheets</p></div><label className="ml-auto flex items-center gap-2 text-xs font-black text-slate-500">Sheet hours <select className="rounded-lg border border-[var(--line)] bg-white px-2 py-2" value={sheet.sheetHours} onChange={e=>setSheet(s=>({...s,sheetHours:Number(e.target.value),rows:s.rows.map(r=>({...r,hours:Number(e.target.value)}))}))}>{Array.from({length:24},(_,i)=><option key={i+1} value={i+1}>{i+1}h</option>)}</select></label></div>
   {tool==="analytics"&&<Analytics rows={sheet.rows} settings={settings}/>} 
   {tool==="settings"&&<Settings settings={settings} setSettings={setSettings} close={()=>setTool(null)}/>} 
   {tool==="calendar"&&<Calendar rows={sheet.rows} close={()=>setTool(null)}/>} 
   {tool==="reports"&&<Reports rows={sheet.rows} settings={settings} close={()=>setTool(null)} download={xlsx}/>} 
   <div className="noprint mb-2 flex items-center gap-1 overflow-x-auto rounded-t-xl border border-b-0 border-[var(--line)] bg-[#e9efec] p-2">{sheets.map((s,i)=><div key={s.id} className={`group flex items-center rounded-lg ${s.id===sheet.id?"bg-white shadow-sm":""}`}><button onClick={()=>setActive(s.id)} className="px-4 py-2 text-xs font-black text-slate-600">{s.name||`Sheet ${i+1}`}</button>{sheets.length>1&&<button onClick={()=>deleteSheet(s.id)} className="pr-2 text-slate-400 hover:text-red-500"><X size={12}/></button>}</div>)}<button disabled={sheets.length>=10} onClick={addSheet} className="rounded-lg bg-white p-2 text-slate-500 disabled:opacity-30"><Plus size={15}/></button></div>
   <div className="overflow-hidden rounded-b-xl border border-[var(--line)] bg-white shadow-sm"><div className="noprint border-b bg-slate-50 px-3 py-2 text-[11px] text-slate-500">Check In / Out is first. A row starts with Check In; after check-in it changes to Check Out. Status can be Off-day or Absent to stop pending check-out. O.T is overtime.</div><Grid rows={sheet.rows} updateRow={updateRow} checkIn={checkIn} checkOut={checkOut} canCheckout={canCheckout} settings={settings} extra={sheet.extra||[]} setSheet={setSheet}/></div>
   <div className="noprint mt-2 grid gap-2 sm:grid-cols-4"><Summary label="Worked Hours" value={total.h.toFixed(2)}/><Summary label="O.T Hours" value={total.ot.toFixed(2)}/><Summary label="Total Earning" value={money(total.earn,settings.currency)}/><Summary label="Sheet Hours" value={`${sheet.sheetHours} h`}/></div>
  </div>
 </main>
}

function TopBar(p){const items=[["analytics",BarChart3,"Analytics"],["calendar",CalendarDays,"Calendar"],["reports",FileText,"Reports"],["settings",Settings2,"Settings"]];return <div className="noprint sticky top-0 z-30 border-b border-[var(--line)] bg-white/95 backdrop-blur"><div className="flex flex-wrap items-center gap-2 px-3 py-3 sm:px-5"><div className="mr-2 font-black">sodah<span className="text-[var(--green)]">.io</span> <span className="text-slate-400">/ Time Sheet</span></div>{items.map(([id,I,t])=><button key={id} onClick={()=>p.setTool(p.tool===id?null:id)} className={`toolbar ${p.tool===id?"active":""}`}><I size={15}/>{t}<ChevronDown size={11}/></button>)}<button className="toolbar" onClick={p.addRow}><Plus size={15}/>Row</button><button className="toolbar" onClick={p.addCol}><Plus size={15}/>Column</button><button className="toolbar" onClick={p.removeRow}><Minus size={15}/>Row</button><button className="toolbar" onClick={p.removeCol}><Minus size={15}/>Column</button><div className="ml-auto flex flex-wrap gap-1"><button className="toolbar" onClick={p.importExcel}><Upload size={15}/>Import</button><button className="toolbar" onClick={p.csv}><Download size={15}/>CSV</button><button className="toolbar" onClick={p.xlsx}><Save size={15}/>Excel</button><button className="toolbar" onClick={()=>window.print()}><Printer size={15}/></button><button className="toolbar primary" disabled={p.count>=10} onClick={p.addSheet}><Plus size={15}/>Sheet</button></div></div></div>}
function Grid({rows,updateRow,checkIn,checkOut,canCheckout,settings,extra,setSheet}){return <div className="sheet-scroll"><table className="excel"><thead><tr><th className="rownum">#</th><th>Check In / Out</th><th>Employee Name</th><th>Employee ID</th><th>Date</th><th>Hours</th><th>O.T</th><th>Hourly Earning</th><th>Check In</th><th>Check Out</th><th>Worked Hours</th><th>O.T Earning</th><th>Total Earning</th><th>Status</th>{extra.map((x,i)=><th key={i}><input className="headinput" value={x} onChange={e=>setSheet(s=>({...s,extra:s.extra.map((z,j)=>j===i?e.target.value:z)}))}/></th>)}</tr></thead><tbody>{rows.map((r,i)=>{const wh=hrs(r.checkIn,r.checkOut),ot=Number(r.otHours||0),rate=Number(r.hourlyEarning||0),otEarn=ot*Number(settings.otRate||0),total=(r.status==="Present"?wh*rate+otEarn:(r.status==="Off-day"?Number(settings.offDayEarning||0):Number(settings.absentEarning||0)));return <tr key={r.id}><td className="rownum">{i+1}</td><td><button onClick={()=>r.checkIn&&!r.checkOut?checkOut(r.id):!r.checkIn&&checkIn(r.id)} disabled={r.status!=="Present"||(r.checkIn&&!r.checkOut&&!canCheckout(r))} className={`action ${r.checkIn&&!r.checkOut?"checkout":""}`}>{r.checkIn&&!r.checkOut?<><LogOut size={14}/>Check Out</>:<><LogIn size={14}/>Check In</>}</button></td><td><input className="cell name" value={r.name} onChange={e=>updateRow(r.id,{name:e.target.value})}/></td><td><input className="cell" value={r.employeeId} onChange={e=>updateRow(r.id,{employeeId:e.target.value})}/></td><td><input className="cell" type="date" value={r.date} onChange={e=>updateRow(r.id,{date:e.target.value})}/></td><td><select className="select" value={r.hours??settings.defaultHours} onChange={e=>updateRow(r.id,{hours:Number(e.target.value)})}>{Array.from({length:24},(_,j)=><option key={j+1} value={j+1}>{j+1}h</option>)}</select></td><td><input className="cell num" type="number" min="0" step="0.25" value={r.otHours??0} onChange={e=>updateRow(r.id,{otHours:Number(e.target.value)})}/></td><td><input className="cell num" type="number" min="0" step="0.01" value={r.hourlyEarning??settings.hourlyEarning} onChange={e=>updateRow(r.id,{hourlyEarning:Number(e.target.value)})}/></td><td className="time">{r.checkIn||"—"}</td><td className="time">{r.checkOut||"—"}</td><td className="calc">{wh.toFixed(2)}</td><td className="calc">{money(otEarn,settings.currency)}</td><td className="calc green">{money(total,settings.currency)}</td><td><select className="select" value={r.status||"Present"} onChange={e=>updateRow(r.id,{status:e.target.value,checkOut:e.target.value==="Present"?r.checkOut:""})}>{STATUSES.map(x=><option key={x}>{x}</option>)}</select></td>{extra.map((_,j)=><td key={j}><input className="cell" value={r["extra"+j]||""} onChange={e=>updateRow(r.id,{["extra"+j]:e.target.value})}/></td>)}</tr>})}</tbody></table></div>}
function Summary({label,value}){return <div className="rounded-xl border border-[var(--line)] bg-white p-3"><div className="text-[10px] font-black uppercase tracking-wider text-slate-400">{label}</div><div className="mt-1 text-lg font-black">{value}</div></div>}
function Analytics({rows,settings}){const h=rows.reduce((a,r)=>a+hrs(r.checkIn,r.checkOut),0),ot=rows.reduce((a,r)=>a+Number(r.otHours||0),0),e=rows.reduce((a,r)=>a+(r.status==="Present"?(hrs(r.checkIn,r.checkOut)*Number(r.hourlyEarning||0)+Number(r.otHours||0)*Number(settings.otRate||0)):0),0);return <div className="mb-3 border border-[var(--line)] bg-[#f9fcfb] p-4"><div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4"><Summary label="Employees" value={new Set(rows.map(r=>r.name).filter(Boolean)).size}/><Summary label="Worked Hours" value={h.toFixed(2)}/><Summary label="O.T Hours" value={ot.toFixed(2)}/><Summary label="Total Earning" value={money(e,settings.currency)}/></div></div>}
function Settings({settings,setSettings,close}){const u=(k,v)=>setSettings(s=>({...s,[k]:v}));return <div className="modalbg"><div className="modal"><div className="modalhead"><div><b>Settings</b><div className="text-xs text-slate-400">All hourly earning and time rules are controlled here.</div></div><button className="iconbtn" onClick={close}><X size={17}/></button></div><div className="grid gap-4 p-5 sm:grid-cols-2">{[["Hourly Earning","hourlyEarning","number"],["Default Hours","defaultHours","number"],["O.T Rate","otRate","number"],["Minimum Check-out Minutes","minimumMinutes","number"],["Work Start","workStart","time"],["Work End / Set Hour","workEnd","time"]].map(([l,k,t])=><label key={k} className="fieldlabel">{l}<input className="field mt-2" type={t} value={settings[k]} onChange={e=>u(k,t==="number"?Number(e.target.value):e.target.value)}/></label>)}<label className="fieldlabel">Mode<select className="field mt-2" value={settings.mode} onChange={e=>u("mode",e.target.value)}><option value="manual">Manual</option><option value="automatic">Automatic</option></select></label><label className="fieldlabel">Currency<select className="field mt-2" value={settings.currency} onChange={e=>u("currency",e.target.value)}><option>USD</option><option>AED</option><option>EUR</option><option>GBP</option><option>GHS</option></select></label><label className="fieldlabel">Off-day Earning<input className="field mt-2" type="number" value={settings.offDayEarning} onChange={e=>u("offDayEarning",Number(e.target.value))}/></label><label className="fieldlabel">Absent Earning<input className="field mt-2" type="number" value={settings.absentEarning} onChange={e=>u("absentEarning",Number(e.target.value))}/></label><label className="flex items-center gap-3 rounded-xl border p-3 text-sm font-bold"><input type="checkbox" checked={settings.autoCheckout} onChange={e=>u("autoCheckout",e.target.checked)}/>Enable automatic check-out</label></div><div className="border-t bg-slate-50 p-5 text-xs leading-5 text-slate-500"><b>Manual:</b> Check-out becomes available after the minimum duration. <b>Automatic:</b> check-out opens only during the final {settings.minimumMinutes} minutes before the configured set hour ({settings.workEnd}).</div></div></div>}
function Calendar({rows,close}){const [date,setDate]=useState(todayISO());const items=rows.filter(r=>r.date===date);return <div className="modalbg"><div className="modal"><div className="modalhead"><b>Calendar</b><button className="iconbtn" onClick={close}><X size={17}/></button></div><div className="p-5"><label className="fieldlabel">Select date<input className="field mt-2" type="date" value={date} onChange={e=>setDate(e.target.value)}/></label><div className="mt-5 space-y-2">{items.length?items.map(r=><div key={r.id} className="rounded-xl border p-3"><b>{r.name||"Employee"}</b><div className="mt-1 text-xs text-slate-500">{r.status} • {r.checkIn||"—"} → {r.checkOut||"—"}</div></div>):<div className="py-8 text-center text-sm text-slate-400">No records for this date.</div>}</div></div></div></div>}
function Reports({rows,settings,close,download}){const [text,setText]=useState("");const h=rows.reduce((a,r)=>a+hrs(r.checkIn,r.checkOut),0),e=rows.reduce((a,r)=>a+(r.status==="Present"?hrs(r.checkIn,r.checkOut)*Number(r.hourlyEarning||0)+Number(r.otHours||0)*Number(settings.otRate||0):0),0);const share=async()=>{const t=`Sodah Time Sheet Report\nEmployees: ${new Set(rows.map(r=>r.name).filter(Boolean)).size}\nWorked Hours: ${h.toFixed(2)}\nTotal Earning: ${money(e,settings.currency)}\n${text}`;if(navigator.share)await navigator.share({title:"Sodah Time Sheet Report",text:t});else await navigator.clipboard?.writeText(t)};return <div className="modalbg"><div className="modal"><div className="modalhead"><div><b>Report</b><div className="text-xs text-slate-400">Generated from the current sheet.</div></div><button className="iconbtn" onClick={close}><X size={17}/></button></div><div className="p-5"><div className="grid gap-3 sm:grid-cols-3"><Summary label="Employees" value={new Set(rows.map(r=>r.name).filter(Boolean)).size}/><Summary label="Worked Hours" value={h.toFixed(2)}/><Summary label="Total Earning" value={money(e,settings.currency)}/></div><label className="fieldlabel mt-5 block">Report text / notes<textarea className="field mt-2 h-28 py-3" value={text} onChange={e=>setText(e.target.value)} placeholder="Add text to the report before improving it with AI..."/></label><div className="mt-4 flex flex-wrap gap-2"><button className="toolbar" onClick={()=>setText(t=>(t? t+"\n\n":"")+"AI improvement: summarize attendance, worked hours, overtime, pending check-outs and earnings for management.")}><Sparkles size={15}/>Improve with AI</button><button className="toolbar" onClick={share}><Share2 size={15}/>Share</button><button className="toolbar" onClick={()=>alert("WhatsApp report handoff is ready to connect to the existing Sodah workflow.")}><Send size={15}/>WhatsApp</button><button className="toolbar" onClick={()=>alert("Email AI report handoff is ready to connect to the existing Sodah email workflow.")}>Email AI</button><button className="toolbar" onClick={()=>window.print()}><Printer size={15}/>Print</button><button className="toolbar primary" onClick={download}><Download size={15}/>Download</button></div></div></div></div>}
